#first event for the "Design Fursona" decision (wbu_design_fursona), an immersive description and a simple continue vs back out option
wbu_religious_decisions.100 = {
    title = wbu_religious_decisions.100.t
	desc = wbu_religious_decisions.100.desc

    #appearance of the event pop-up
    type = character_event #standard pop-up event, rather than a letter or a duel
	hidden = no
    theme = faith
    override_background = study
	left_portrait = {
		character = root
		animation = writing #looks close enough to drawing
	}

    #continue the decision
    option = {
        name = wbu_religious_decisions.100.go
        trigger_event = wbu_religious_decisions.101
    }

    #back out of the decision
    option = {
        #name = wbu_religious_decisions.100.optout
        name = religious_decision.0101.optout #use existing localization
        custom_tooltip = wbu_religious_decisions.100.optout_cooldown_warning_tooltip
        ai_chance = { factor = 0 } #ai will never choose this option
    }
}

#reworked third event for "Design Fursona", where the player chooses to align with or to go against their personality traits
wbu_religious_decisions.102 = {
    title = wbu_religious_decisions.102.t
    desc = wbu_religious_decisions.102.desc

    #appearance of the event pop-up
    type = character_event #standard pop-up event, rather than a letter or a duel
    hidden = no
    theme = faith
    override_background = study
    left_portrait = {
        character = root
        animation = writing #looks close enough to drawing
    }

    #options
    option = {
        name = wbu_religious_decisions.102.align
        set_variable = {
            name = wbu_religious_decisions_102_choice
            value = flag:align
        }
        custom_tooltip = wbu_religious_decisions.102.align_tooltip
    }
    option = {
        name = wbu_religious_decisions.102.oppose
        #adds a flag to the character (ROOT)
        set_variable = {
            name = wbu_religious_decisions_102_choice
            value = flag:oppose
        }
        custom_tooltip = wbu_religious_decisions.102.oppose_tooltip
    }
    after = {
        trigger_event = wbu_religious_decisions.103
    }
}
