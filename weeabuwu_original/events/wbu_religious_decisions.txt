#First event for the "Design Fursona" decision (wbu_design_fursona), an immersive description and a simple continue vs back out option
wbu_religious_decisions.100 = {
    title = wbu_religious_decisions.100.t
	desc = wbu_religious_decisions.100.desc

    #appearance of the event pop-up
    type = character_event #standard pop-up event, rather than a letter or a duel
	hidden = no
    theme = faith
    override_background = study
	left_portrait = {
		character = root
		animation = writing #looks close enough to drawing
	}

    #continue the decision
    option = {
        name = wbu_religious_decisions.100.go
        trigger_event = wbu_religious_decisions.101
    }

    #back out of the decision
    option = {
        #name = wbu_religious_decisions.100.optout
        name = religious_decision.0101.optout #use existing localization
        ai_chance = { factor = 0 } #ai will never choose this option
    }
}

#second event for "Design Fursona", where the player chooses a color aspect of their fursona
wbu_religious_decisions.101 = {
    title = wbu_religious_decisions.101.t
    desc = wbu_religious_decisions.101.desc

    #appearance of the event pop-up
    type = character_event #standard pop-up event, rather than a letter or a duel
	hidden = no
    theme = faith
    override_background = study
	left_portrait = {
		character = root
		animation = writing #looks close enough to drawing
	}

    #options
    #natural / earthy colors
    # WHAT THE FUCK IS A SCRIPTED EFFECT?!?!?!?
    option = {
        name = wbu_religious_decisions.101.earth
        wbu_set_fursona_color_effect = { FURSONA_COLOR = wbu_fursona_color_earth }
    }

    #white / light colors
    option = {
        name = wbu_religious_decisions.101.snow
        wbu_set_fursona_color_effect = { FURSONA_COLOR = wbu_fursona_color_snow }
    }

    #dark / shadowy colors
    option = {
        name = wbu_religious_decisions.101.night
        wbu_set_fursona_color_effect = { FURSONA_COLOR = wbu_fursona_color_night }
    }

    #sparkledog colors
    option = {
        name = wbu_religious_decisions.101.sparkle
        wbu_set_fursona_color_effect = { FURSONA_COLOR = wbu_fursona_color_sparkle }
    }

    #fire / ruddy colors
    option = {
        name = wbu_religious_decisions.101.fire
        wbu_set_fursona_color_effect = { FURSONA_COLOR = wbu_fursona_color_fire }
    }

    #after is a block of effects that happen after any option is chosen. It's counterpart is immediate.
    after = {
        add_piety = 100 #reward for completing this phase
        trigger_event = wbu_religious_decisions.102
    }
}
