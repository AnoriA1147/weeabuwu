#This is basically a manager system that automatically removes modifiers before adding new ones.
#heavily copied from base Bakhti scripted effects
wbu_set_fursona_color_effect = {
    if = {
        limit = { faith = faith:furry }
        if = {
			limit = {
				has_character_modifier = wbu_fursona_color_earth
			}
			remove_character_modifier = wbu_fursona_color_earth
			#add_piety = massive_piety_loss
            #removed piety loss. changing fursonas is frequent in lore.
		}
        else_if = {
			limit = {
				has_character_modifier = wbu_fursona_color_snow
			}
			remove_character_modifier = wbu_fursona_color_snow
		}
        else_if = {
			limit = {
				has_character_modifier = wbu_fursona_color_night
			}
			remove_character_modifier = wbu_fursona_color_night
		}
        else_if = {
			limit = {
				has_character_modifier = wbu_fursona_color_sparkle
			}
			remove_character_modifier = wbu_fursona_color_sparkle
		}
        else_if = {
			limit = {
				has_character_modifier = wbu_fursona_color_fire
			}
			remove_character_modifier = wbu_fursona_color_fire
		}
    }
    else = {
		# If no current bhakti, cost is reduced.
		add_piety = medium_piety_loss
	}

	add_character_modifier = $FURSONA_COLOR$
}
